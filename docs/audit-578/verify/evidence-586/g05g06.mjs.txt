const clerk = async (method, route, body) => {
  const r = await fetch(`https://api.clerk.com/v1${route}`, {
    method, headers: { Authorization: `Bearer ${SECRET}`, "Content-Type": "application/json" },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  const data = await r.json().catch(() => null);
  if (!r.ok) throw new Error(`Clerk ${method} ${route} -> ${r.status} ${JSON.stringify(data?.errors ?? data)}`);
  return data;
};
const audit = async (method, route, body) => {
  const r = await fetch(`${BASE}${route}`, {
    method, headers: { "X-Admin-Audit-Key": AUDIT, Origin: BASE, Accept: "application/json", ...(body ? { "Content-Type": "application/json" } : {}) },
    body: body ? JSON.stringify(body) : undefined,
  });
  const data = await r.json().catch(() => null);
  if (!r.ok) throw new Error(`${method} ${route} -> ${r.status} ${JSON.stringify(data)}`);
  return data;
};

async function signIn(page) {
  await page.goto(`${BASE}/sign-in`, { waitUntil: "domcontentloaded", timeout: 60_000 });

  await page.keyboard.press("Enter");
  await page.locator('input[name="password"]').fill(password);
  await page.keyboard.press("Enter");
  const otp = page.locator('input[aria-label="Enter verification code"]');
  const deadline = Date.now() + 90_000;
  while (Date.now() < deadline) {
    if (await page.evaluate(() => Boolean(window.Clerk?.user)).catch(() => false)) return;
    if (await otp.isVisible().catch(() => false)) { await otp.click(); await page.keyboard.type("424242", { delay: 120 }); }
    await page.waitForTimeout(500);
  }
  throw new Error("sign-in never completed");
}
async function authUser(page) {
  return page.evaluate(async () => (await fetch("/api/auth/user", { credentials: "include" })).json().catch(() => null));
}
